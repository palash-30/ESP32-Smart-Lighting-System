const SHEET_NAME = "Log_Sheet";

// =====================================================
// Main GET Function
// =====================================================

function doGet(e) {

  const params = e.parameter;

  // ===================================================
  // 5-Minute Email Report
  // ===================================================

  if (params.report === "5min") {

    sendFiveMinuteEmail(params);

    return ContentService
      .createTextOutput(
        JSON.stringify({
          result: "Email Sent"
        })
      )
      .setMimeType(ContentService.MimeType.JSON);
  }

  // ===================================================
  // Normal Event Logging
  // ===================================================

  const sheet = SpreadsheetApp
    .getActiveSpreadsheet()
    .getSheetByName(SHEET_NAME);

  const date = params.date || "";
  const time = params.time || "";
  const event = params.event || "";
  const motion = params.motion || "";
  const light = params.light || "";
  const mode = params.mode || "";
  const pir = params.pir || "";

  // ===================================================
  // Generate Serial Number
  // ===================================================

  const lastRow = sheet.getLastRow();

  let serialNumber = 1;

  if (lastRow >= 2) {

    const firstSerialNumber =
      Number(sheet.getRange(2, 1).getValue());

    if (!isNaN(firstSerialNumber) && firstSerialNumber > 0) {
      serialNumber = firstSerialNumber + 1;
    }
  }

  // ===================================================
  // Insert New Row at Row 2
  // ===================================================

  sheet.insertRowBefore(2);

  // ===================================================
  // Write New Entry
  // ===================================================

  sheet.getRange(2, 1, 1, 8).setValues([[
    serialNumber,
    date,
    time,
    event,
    motion,
    light,
    mode,
    pir
  ]]);

  // ===================================================
  // Response
  // ===================================================

  return ContentService
    .createTextOutput(
      JSON.stringify({
        result: "Success",
        serialNumber: serialNumber,
        row: 2
      })
    )
    .setMimeType(ContentService.MimeType.JSON);
}


// =====================================================
// 5-Minute Email
// =====================================================

function sendFiveMinuteEmail(params) {

  const recipients = [
  "YOUR_EMAIL_1",
  "YOUR_EMAIL_2",
  "YOUR_EMAIL_3"
];

  const date = params.date || "";
  const time = params.time || "";

  const motion =
    params.motion || "0";

  const totalMotion =
    params.totalMotion || "0";

  const light =
    params.light || "UNKNOWN";

  const mode =
    params.mode || "UNKNOWN";

  const ip =
    params.ip || "UNKNOWN";

  const subject =
    "ESP32 Smart Lighting - 5 Minute Report";

  const body =
    "ESP32 SMART LIGHTING SYSTEM\n\n" +

    "5-Minute Activity Report\n" +
    "==============================\n\n" +

    "Date: " + date + "\n" +
    "Time: " + time + "\n\n" +

    "Motion Detected in Last 5 Minutes: "
    + motion + "\n" +

    "Total Motion Count: "
    + totalMotion + "\n\n" +

    "Current Light Status: "
    + light + "\n" +

    "Control Mode: "
    + mode + "\n\n" +

    "ESP32 IP Address: "
    + ip + "\n\n" +

    "==============================\n" +

    "Automated report generated by\n" +
    "ESP32 Smart Lighting System.";

  MailApp.sendEmail({
    to: recipients.join(","),
    subject: subject,
    body: body
  });
}
